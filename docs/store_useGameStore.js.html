import { create } from "zustand";
import { persist } from "zustand/middleware";
/** @module Store */
/**
 * @typedef {Object} GameResult
 * @property {number} id - Унікальний ідентифікатор результату.
 * @property {string} player - Ім'я гравця.
 * @property {number} moves - Кількість ходів, за які гравець завершив гру.
 * @property {string} date - Дата та час завершення гри у форматі рядка.
 */

/**
 * Zustand store для гри "Memory Game".
 *
 * Відповідає за:
 * - Зберігання кількості пар карток (`cardCount`)
 * - Історії результатів гри (`history`)
 * - Методи для оновлення кількості карток та історії
 *
 * @typedef {Object} GameStore
 * @property {number} cardCount - Кількість пар карток для гри.
 * @property {Array<GameResult>} history - Історія останніх результатів (максимум 10).
 * @property {function(number): void} setCardCount - Встановлює нову кількість карток.
 * @property {function(GameResult): void} addResult - Додає новий результат до історії.
 * @property {function(): void} clearHistory - Очищає історію результатів.
 */

/**
 * Глобальний store гри з використанням Zustand та persist.
 * Дані зберігаються в localStorage під ключем "game-storage".
 *
 * @returns {GameStore} Об'єкт зі станом гри та методами.
 */
export const useGameStore = create(
  persist(
    (set) => ({
      cardCount: Number(localStorage.getItem("gamePairs")) || 8,

      history: [],

      setCardCount: (count) => set({ cardCount: count }),

      addResult: (result) =>
        set((state) => ({
          history: [result, ...state.history].slice(0, 10),
        })),

      clearHistory: () => set({ history: [] }),
    }),
    {
      name: "game-storage",
    },
  ),
);
